# Flight check (ctest `flight`): from the meadow's rim the player runs, jumps, flaps once and glides to the stepping
# isle, gathering the light over the gap (a third mark on the cape); a flap there spends a mark and the lantern warms
# it back; then two flaps and a long glide carry it across to the spirit isle through the next light.
#   cloudsong --script scenarios/flight.txt

wait 0.5
teleport -22.2 3 28.4 -0.664
wait 0.8
expect grounded == 1
expect energy == 2
move 0 1
wait 1.5
jump down
wait 0.3
jump up
wait 0.05
jump down
until glide > 0.5 2
# The cape's ribbons while gliding; the streams of light and the meadow's points are drawn.
expect wing_trail == 1
expect trails > 2
expect particles > 100
until grounded == 1 4
expect lights == 1
expect max_energy == 3
expect x < -29
expect z > 38
move 0 0
jump up
wait 0.5
expect wing_trail == 0
# A flap in place spends a mark; the lantern warms it back.
jump down
wait 0.3
jump up
wait 0.05
jump down
wait 0.03
expect energy < 2.1
wait 0.07
jump up
until grounded == 1 3
expect warm == 1
until energy >= 3 2

# Toward the spirit isle.
teleport -36 4 46 -0.534
wait 0.6
move 0 1
wait 0.6
jump down
wait 0.3
jump up
wait 0.05
jump down
wait 0.5
jump up
wait 0.05
jump down
until lights == 2 4
until grounded == 1 6
expect x < -52
expect z > 78
quit
