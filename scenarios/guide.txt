# Guide check (ctest `guide`): the call's guide. None before a call; after one, a few heads rise beside the figure,
# fly to the nearest cold candle (the pond's), linger there and fade; a call while it flies sends the same guide on
# (no more ribbons); with the goal at hand it circles the figure once; with every candle lit it flies to the updraft.
# The particles and trails per picture setting are logged (the actual values of the `>= 0` expectations).
#   cloudsong --script scenarios/guide.txt

until nav_ready == 1 10
quality high
teleport 2 0.8 -36 0
wait 1.5
log "high: before a call"
# Other residents have their own eye trails; measure only the added guide ribbons.
mark trails
expect guide == 0
expect d_trails == 0
expect particles >= 0
expect particle_systems >= 0
call
wait 0.5
expect guide == 1
expect guide_goal == 0
expect d_trails >= 2
mark guide_dist
wait 2.5
log "high: the guide in flight"
expect d_guide_dist < -8
expect particles >= 0
expect trails >= 0
until guide_dist < 2 8
wait 1
expect guide == 1
until guide == 0 8
wait 0.3
expect d_trails == 0

# A call while it flies sends it on: still one guide's ribbons (3 heads with their bands on this setting).
call
wait 1
call
wait 0.5
expect guide == 1
expect d_trails >= 2
expect d_trails <= 6
until guide == 0 20

# The goal at hand (the pond's candle 5 m away, out of the call's reach): once round the figure.
teleport -25.4 3 -15.2 1.57
until grounded == 1 4
wait 0.5
call
wait 0.5
expect candles == 0
expect guide == 1
expect guide_goal == -1
expect guide_dist == 0
until guide == 0 10

# Every candle lit: to the open updraft.
light 0
light 1
light 2
teleport 2 0.8 -36 0
wait 0.5
expect updraft == 1
call
wait 0.5
expect guide == 1
expect guide_goal == 3
mark guide_dist
wait 2
expect d_guide_dist < -5
until guide == 0 30

quality medium
wait 1
log "medium: before a call"
mark trails
expect d_trails == 0
expect particles >= 0
call
wait 3
log "medium: the guide in flight"
expect guide == 1
expect particles >= 0
expect trails >= 0
until guide == 0 30

quality low
wait 1
log "low: before a call"
mark trails
expect d_trails == 0
expect particles >= 0
call
wait 3
log "low: the guide in flight"
expect guide == 1
expect particles >= 0
expect trails >= 0
until guide == 0 30
quit
