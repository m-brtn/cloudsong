# The solver runs at fixed 120 Hz; this deterministic scenario checks every
# frame's peak, not one lucky resting frame. The idle window covers the full
# 8.333 s imported loop. Captures are outside any real-time performance region.
screen play
day_cycle off
time_of_day 12
until hero_ready == 1 10
teleport 2 2 -36 0
move 0 0
jump up
wait .1
until grounded == 1 4
wait 3
garment_audit_reset
wait 9
expect grounded == 1
expect hero_pose_valid == 1
expect hero_cape_motion_peak < .1
expect hero_hood_motion_peak < .15
expect hero_cape_jitter_peak < .04
expect hero_hood_jitter_peak < .07
view 2 1.8 -38.2 2 1.45 -36
capture garments/rest.png
view off
camera 0 .12 3
# Inertia remains active through locomotion instead of freezing the garments.
move 0 1
wait .6
garment_audit_reset
wait .5
expect hero_pose_valid == 1
expect hero_cape_motion_peak > .02
expect hero_hood_motion_peak > .005
move 0 0
wait 3
garment_audit_reset
wait 2
expect grounded == 1
expect hero_pose_valid == 1
expect hero_cape_motion_peak < .1
expect hero_hood_motion_peak < .15
expect hero_cape_jitter_peak < .04
expect hero_hood_jitter_peak < .07
capture garments/settled.png
# Flight keeps the same finite cage/palette after a reset and a large pose change.
teleport -40 30 70 0
jump down
wait .9
expect glide > .5
expect hero_pose_valid == 1
capture garments/glide.png
jump up
quit
