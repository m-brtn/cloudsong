# The title, the settings, the start, resting (the pause menu) and the touch controls under scripted fingers
# (ctest `front`). Scripted runs start playing; `screen` shows the title or the menus as a player would see them.
#   cloudsong --script scenarios/front.txt

screen title
wait 0.5
expect screen == 0
# Space begins from the title.
tap Space
wait 0.1
expect screen == 1
wait 1.0

# Escape rests and resumes; resting holds the player still.
tap Escape
wait 0.1
expect screen == 2
mark z
down W
wait 0.4
up W
expect d_z < 0.01
tap Escape
wait 0.1
expect screen == 1

# A held gamepad confirmation must not leak through Resume into jump/glide.
gamepad attach
gamepad button start tap
wait 0.3
expect screen == 2
# Establish navigation focus on Resume (the pointer may previously have owned focus).
gamepad button dpdown tap
wait 0.1
gamepad button dpup tap
wait 0.1
gamepad button a down
wait 0.15
expect screen == 1
expect grounded == 1
wait 0.4
expect grounded == 1
expect glide == 0
gamepad button a up
wait 0.1
# A genuinely new press still jumps.
gamepad button a down
wait 0.15
expect grounded == 0
gamepad button a up
until grounded == 1 3
gamepad detach
wait 0.1

# The quality's render settings.
quality low
expect render_scale < 0.71
quality high
expect render_scale > 0.99

# Touch: a thumb dragged up on the left walks forward, the arrow jumps, the pause button rests.
touch_controls on
wait 0.2
mark z
touch 15% 80% 0 -70 70
wait 0.1
expect d_z > 1.0
wait 0.5
mark y
touch 1194 628
wait 0.25
expect d_y > 0.4
wait 1.2
expect grounded == 1
touch 1246 34
wait 0.1
expect screen == 2

# The settings, play with the touch controls and the debug window, the title: no two items of one window shared an
# ID in any frame of the run (the engine counts them in scripted runs).
screen settings
wait 0.2
debug_ui on
screen play
wait 0.2
screen title
wait 0.2
expect imgui_id_conflicts == 0
quit
